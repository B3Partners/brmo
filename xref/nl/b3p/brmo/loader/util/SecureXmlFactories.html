3   import javax.xml.XMLConstants;
4   import javax.xml.parsers.DocumentBuilderFactory;
5   import javax.xml.parsers.ParserConfigurationException;
6   import javax.xml.stream.XMLInputFactory;
7   import javax.xml.transform.TransformerConfigurationException;
8   import javax.xml.transform.TransformerFactory;
9   import org.apache.commons.xml.secure.SecureDocumentBuilderFactory;
10  import org.apache.commons.xml.secure.SecureTransformerFactory;
11  import org.apache.commons.xml.secure.SecureXMLInputFactory;
12  
13  /**
14   * Factories for XML processing which are hardened against XXE attacks using Apache Commons Secure
15   * XML, which resolves external references to empty content. On top of that the {@link
16   * DocumentBuilderFactory} and {@link XMLInputFactory} reject documents with a DOCTYPE, so a
17   * malicious document fails instead of being silently processed.
18   */
19  public final class SecureXmlFactories {
20    private SecureXmlFactories() {}
21  
22    /**
23     * Create a {@link TransformerFactory} which resolves external entities, DTDs, stylesheets and
24     * documents to empty content and has secure processing enabled (entity expansion limits, no
25     * extension functions). Stylesheets loaded via a {@link javax.xml.transform.URIResolver} set on
26     * the factory are still allowed.
27     *
28     * @return hardened transformer factory
29     * @throws TransformerConfigurationException if the implementation does not support secure
30     *     processing
31     */
32    public static TransformerFactory newTransformerFactory()
33        throws TransformerConfigurationException {
34      TransformerFactory tf = SecureTransformerFactory.newInstance();
35      tf.setFeature(XMLConstants.FEATURE_SECURE_PROCESSING, true);
36      return tf;
37    }
38  
39    /**
40     * Create a {@link DocumentBuilderFactory} which rejects documents with a DOCTYPE declaration, so
41     * no external entities, external DTDs or entity expansion (billion laughs) are possible.
42     *
43     * @return hardened document builder factory
44     * @throws ParserConfigurationException if the implementation does not support the required
45     *     features
46     */
47    public static DocumentBuilderFactory newDocumentBuilderFactory()
48        throws ParserConfigurationException {
49      DocumentBuilderFactory dbf = SecureDocumentBuilderFactory.newInstance();
50      dbf.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
51      return dbf;
52    }
53  
54    /**
55     * Create a {@link XMLInputFactory} which does not support DTDs, so no external entities, external
56     * DTDs or entity expansion (billion laughs) are possible.
57     *
58     * @return hardened StAX input factory
59     */
60    public static XMLInputFactory newXMLInputFactory() {
61      XMLInputFactory xif = SecureXMLInputFactory.newInstance();
62      xif.setProperty(XMLInputFactory.SUPPORT_DTD, Boolean.FALSE);
63      return xif;
64    }
65  }

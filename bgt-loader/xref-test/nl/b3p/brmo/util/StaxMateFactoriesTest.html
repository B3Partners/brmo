8   package nl.b3p.brmo.util;
9   
10  import static org.junit.jupiter.api.Assertions.*;
11  
12  import java.io.StringReader;
13  import org.codehaus.stax2.ri.Stax2ReaderAdapter;
14  import org.junit.jupiter.api.Test;
15  
16  class StaxMateFactoriesTest {
17    private static final String FACTORY_PROPERTY = "javax.xml.stream.XMLInputFactory";
18  
19    @Test
20    void usesNativeStax2Reader() throws Exception {
21      assertFalse(
22          StaxMateFactories.newSMInputFactory()
23                  .rootElementCursor(new StringReader("<r/>"))
24                  .getStreamReader()
25              instanceof Stax2ReaderAdapter);
26    }
27  
28    @Test
29    void rejectsJdkStaxImplementation() {
30      String previous = System.getProperty(FACTORY_PROPERTY);
31      System.setProperty(FACTORY_PROPERTY, "com.sun.xml.internal.stream.XMLInputFactoryImpl");
32      try {
33        IllegalStateException e =
34            assertThrows(IllegalStateException.class, StaxMateFactories::newSMInputFactory);
35        assertTrue(e.getMessage().contains("does not implement Stax2"), e.getMessage());
36      } finally {
37        if (previous == null) {
38          System.clearProperty(FACTORY_PROPERTY);
39        } else {
40          System.setProperty(FACTORY_PROPERTY, previous);
41        }
42      }
43    }
44  }

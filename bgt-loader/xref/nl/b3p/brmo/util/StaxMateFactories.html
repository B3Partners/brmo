7   package nl.b3p.brmo.util;
8   
9   import java.io.StringReader;
10  import javax.xml.stream.XMLInputFactory;
11  import javax.xml.stream.XMLStreamException;
12  import javax.xml.stream.XMLStreamReader;
13  import org.apache.commons.xml.secure.SecureXMLInputFactory;
14  import org.codehaus.stax2.XMLStreamReader2;
15  import org.codehaus.staxmate.SMInputFactory;
16  
17  /** Factory for StaxMate input factories for parsing GML with StaxMate and GeoTools. */
18  public final class StaxMateFactories {
19    private StaxMateFactories() {}
20  
21    /**
22     * Create a StaxMate input factory using a StAX implementation which is secured against XXE by
23     * Apache Commons Secure XML, coalesces characters and does not support DTDs.
24     *
25     * <p>The StAX implementation must natively implement Stax2, such as Woodstox. For other
26     * implementations (such as the JDK default) StaxMate wraps the reader in a {@link
27     * org.codehaus.stax2.ri.Stax2ReaderAdapter} which does not track the element depth for {@link
28     * XMLStreamReader#nextTag()}. The GeoTools XmlStreamGeometryReader uses that method, so after
29     * reading some geometries (such as curves) StaxMate cursors silently skip the elements after it.
30     *
31     * <p>The GeoTools geometry reader reads from {@link
32     * org.codehaus.staxmate.in.SMInputCursor#getStreamReader()} directly, bypassing the cursors.
33     * StaxMate allows this but discourages it, because advancing the reader directly can desync the
34     * cursors (see <a href="https://github.com/FasterXML/StaxMate/issues/2">StaxMate issue #2</a>).
35     * As of StaxMate 2.4.2 this works because cursors resync using {@code getDepth()} of the reader,
36     * which is only correct for all methods with a native Stax2 implementation, and because the
37     * geometry is read within the element of a child cursor which is not used afterward.
38     *
39     * <p>If a future StaxMate version resolves issue #2 (for instance by wrapping the returned reader
40     * to keep the cursors synchronized), this check may no longer be needed. Note that StaxMate may
41     * be upgraded by automatic dependency updates without anyone checking whether that issue has been
42     * resolved, so keep the check unless it has been verified to be unnecessary.
43     *
44     * @return the StaxMate input factory
45     * @throws IllegalStateException if the StAX implementation does not natively implement Stax2
46     */
47    public static SMInputFactory newSMInputFactory() {
48      final XMLInputFactory xmlInputFactory = SecureXMLInputFactory.newFactory();
49      xmlInputFactory.setProperty(XMLInputFactory.IS_COALESCING, Boolean.TRUE); // Coalesce characters
50      xmlInputFactory.setProperty(
51          XMLInputFactory.SUPPORT_DTD,
52          Boolean.FALSE); // No XML entity expansions or external entities
53  
54      try {
55        XMLStreamReader reader = xmlInputFactory.createXMLStreamReader(new StringReader("<r/>"));
56        if (!(reader instanceof XMLStreamReader2)) {
57          throw new IllegalStateException(
58              "StAX implementation "
59                  + reader.getClass().getName()
60                  + " does not implement Stax2, use Woodstox");
61        }
62        reader.close();
63      } catch (XMLStreamException e) {
64        throw new IllegalStateException("Error checking StAX implementation", e);
65      }
66  
67      return new SMInputFactory(xmlInputFactory);
68    }
69  }
